package edu.odu.cs.cs350;

import java.time.LocalDate;
import java.time.temporal.ChronoUnit;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;

//TODO: This class, and others along the Semester-Snapshot-Course-Offering-Section hierarchy could use more Java checklist methods, particularly equals(), compareto() and toString()
/**
 * A Semester; for our purposes it is composed of a title, code, pre-registration and add-deadline dates, as well as several snapshots of enrollment relevent to the semester.
 */
public class Semester {

    /** Human Readable semester title (e.g Spring 2021). */
    private final String semesterTitle;
    /** The actual semestercode, needed for validation in main. */
    private final String semesterCode;

    /** The date of preregistration. */
    private final LocalDate dateOfPreregistration;
    /** The addDeadlineDate. */
    private final LocalDate addDeadline;
    /** This is not simply addDeadline, as it could be either the deadline from dates.txt, or the user supplied cutoff. */
    private final LocalDate effectiveDeadline;
    /** The snapshots that we have for this semester. */
    private final ArrayList<EnrollmentSnapshot> enrollmentSnapshots;

    /**
     * Constructor for semester.
     * 
     * @param snapshots the snapshots of the semester in questions
     * @param preregistration the pre-registration date
     * @param effectiveDeadline the effective deadline; relevent for the last semester where user supplied arg may preceed the addDeadline
     * @param semesterCode designation for the semester, translated into a tile
     * @param addDeadline the add deadline date
     */
    public Semester(List<EnrollmentSnapshot> snapshots, LocalDate preregistration, LocalDate effectiveDeadline, String semesterCode, LocalDate addDeadline)
    {
        enrollmentSnapshots = enforceNonEmptySortedSnapshots(snapshots);
        dateOfPreregistration = preregistration;
        this.effectiveDeadline = effectiveDeadline;
        checkDates(preregistration, effectiveDeadline);
        semesterTitle =  parseSemesterCodeToTitle(semesterCode);
        this.semesterCode = semesterCode;
        this.addDeadline = addDeadline;

    }

    /**
     * Invariant mechanism to force legitimate dates.
     * @param dateOfPreregistration pre-registration date
     * @param deadlineDate add-deadline date
     * @throws IllegalArgumentException if pre-registration is after add deadline 
     */
    public static void checkDates(LocalDate dateOfPreregistration, LocalDate deadlineDate ) throws IllegalArgumentException
    {
        if(dateOfPreregistration.compareTo(deadlineDate) > 0 )
        {
            throw new IllegalArgumentException("Error with provided dates: add deadline is before preregistration");
        }
    }

    /**
     * Invariant mechanism to force at least 2 snapshots per semester.
     * And order snapshots by date (most recent = last).
     * 
     * @param snapshotList the snapshots to check
     * @return the snapshots, if legitimate
     * @throws IllegalArgumentException if size < 2
     */
    public static ArrayList<EnrollmentSnapshot> enforceNonEmptySortedSnapshots(List<EnrollmentSnapshot> snapshotList)
    {

        if(snapshotList.size() < 2)
        {
            throw new IllegalArgumentException("A semester must have at least two snapshots to be considered.");
        }

        var nonEmptySnaps =  new ArrayList<>(snapshotList);
        nonEmptySnaps.sort(Comparator.comparing(EnrollmentSnapshot::getSnapShotDateValue));
        return nonEmptySnaps;
    }
    
    /**
     * Derives the semester title from the semester code.
     * 
     * @param semesterCode semester code (e.g 202010)
     * @return semester title (e.g Spring 2020)
     */
    public static String parseSemesterCodeToTitle(String semesterCode)
    {
        String semesterPeriod = semesterCode.substring(semesterCode.length()-2, semesterCode.length());

        switch (semesterPeriod) {
            case "10" -> semesterPeriod = "Fall ";
            case "20" -> semesterPeriod = "Spring ";
            case "30" -> semesterPeriod = "Summer ";
            default -> throw new IllegalArgumentException("Semester codes must end with 10, 20, or 30");
        }


        return semesterPeriod +  semesterCode.substring(0, semesterCode.length()-2);
    }

    /**
     * Calculates % of enrollment period passed for summary projection report header.
     * 
     * @param preregistration date of pre-registration
     * @param addDeadline date of add-deadline
     * @param snapDate date of snapshot
     * @return the % (0-100) of time passed from pre-registration to the add-deadline as of the snapshot date
     */
    public static long enrollmentPeriodPassed(LocalDate preregistration, LocalDate addDeadline, LocalDate snapDate){

        //finds the difference between entered dates to calculate total enrollment period time and the number of days into it
        long percent = 100;
        long totalPeriod = ChronoUnit.DAYS.between(preregistration, addDeadline);
        long daysPassed = ChronoUnit.DAYS.between(preregistration, snapDate);

        //throws exception if add deadline is somehow before pregristration dates
        if(totalPeriod < 0){
            throw new IllegalArgumentException("Error with provided dates: add deadline is before preregistration");
        }

        //returns 100% if add deadline is day of preregistration
        if (totalPeriod == 0){ return percent;}
        
        //calculation for % of enrollment period passed
        percent = daysPassed*100/totalPeriod;
        
        //sets anything < 0 to 0% and > 100 to 100% and returns % value
        return Math.max(0, Math.min(100, percent));
    }

    /**
     * 
     * @return the human readable semester title
     */
    public String getSemesterTitle() {
        return semesterTitle;
    }

    /**
     * 
     * @return the pre-registration date
     */
    public LocalDate findPreregistrationDate() {
        return dateOfPreregistration;
    }

    /**
     * 
     * @return the effective deadline date
     */
    public LocalDate getEffectiveDeadline() {
        return effectiveDeadline;
    }

    /**
     * 
     * @return the enrollment snapshots
     */
    public ArrayList<EnrollmentSnapshot> getEnrollmentSnapshots() {
        return enrollmentSnapshots;
    }

    /**
     * 
     * @return the add deadline
     */
    public LocalDate getAddDeadline() {
        return addDeadline;
    }

    /**
     * 
     * @return the semester code
     */
    public String getSemesterCode() {
        return semesterCode;
    }
}
