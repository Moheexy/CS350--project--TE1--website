package edu.odu.cs.cs350;

/**
 * 
 * A course Section. Composed of
 * @param CRN unique identifier for each section
 * @param subject Identifies the department offering the course
 * @param CourseNumber Course number (e.g CS - 350)
 * @param CrossListCap (cross-list cap) The maximum number of students that can enroll in this section.
 * @param ENR Number of students enrolled in this section
 * @param LINK labs or recitations associated lecture
 * @param crossListGroup (cross-list group) links section(s) to an offering
 * @param overallEnrollmentCap maximum number of students that can enroll in the offering to which this section belong
 * @param overallEnrollment total number of students enrolled in the offering to which this section belongs
 */
public record Section(
     int CRN,
     String subject,
     String CourseNumber,
     int CrossListCap,
     int ENR,
     String LINK,
     String crossListGroup,
     int overallEnrollmentCap,
     int overallEnrollment
    ){

    /**
     * Section record.
     * 
     * @param CRN unique identifier for each section
     * @param subject Identifies the department offering the course
     * @param CourseNumber Course number (e.g CS - 350)
     * @param CrossListCap (cross-list cap) The maximum number of students that can enroll in this section.
     * @param ENR Number of students enrolled in this section
     * @param LINK labs or recitations associated lecture
     * @param crossListGroup (cross-list group) links section(s) to an offering
     * @param overallEnrollmentCap maximum number of students that can enroll in the offering to which this section belong
     * @param overallEnrollment total number of students enrolled in the offering to which this section belongs
     */
    public Section{

        checkCRN(CRN);

        checkCrossListCap(CrossListCap);

        enforceEnrollmentPositive(ENR);


        checkLinkMatchesInvariant(LINK);
                
        LINK = assignNullLinkToNone(LINK);

        crossListGroup = assignCrossListGroupToNone(crossListGroup);

        checkOverallEnrollmentCap(overallEnrollmentCap);

        checkOverallEnrollment(overallEnrollment);

        //This invariant does not hold in real data
        // if(ENR > XLSTCAP)
        // {
        //     throw new IllegalArgumentException("Enrollment cannot exceed cap for section.");
        // }
        //This invariant does not hold in real data
        // if(overallEnrollment > overallEnrollmentCap)
        // {
        //     throw new IllegalArgumentException("Overall enrollment must be less than overall cap.");
        // }


    }

    private void checkOverallEnrollment(int overallEnrollment1) throws IllegalArgumentException {
        if (overallEnrollment1 < 0) {
            throw new IllegalArgumentException("Overall enrollment must be non-negative.");
        }
    }

    private void checkOverallEnrollmentCap(int overallEnrollmentCap1) throws IllegalArgumentException {
        if (overallEnrollmentCap1 < 0) {
            throw new IllegalArgumentException("Enrollment cap must be non-negative.");
        }
    }

    private String assignCrossListGroupToNone(String crossListGroup1) {
        if (crossListGroup1 == null || crossListGroup1.isBlank()) {
            return "None";
        }
        return crossListGroup1;
    }

    private String assignNullLinkToNone(String LINK1) {
        if (LINK1 == null || LINK1.isBlank()) {
            return "None";
        }
        return LINK1;
    }

    private void checkLinkMatchesInvariant(String LINK1) throws IllegalArgumentException {
        if (LINK1 != null && !LINK1.isBlank()) {
            char LinkEnd = LINK1.charAt(LINK1.length() - 1);
            if(LinkEnd != '1' && LinkEnd != '2' && LinkEnd != '3')
            {
                throw new IllegalArgumentException("If present, linked sections must be specified with 1,2, or 3 as final character.");
            }
        }
    }

    private void enforceEnrollmentPositive(int ENR1) throws IllegalArgumentException {
        if (ENR1 < 0) {
            throw new IllegalArgumentException("Enrollment must be nonnegative.");
        }
    }

    private void checkCrossListCap(int CrossListCapToCheck) throws IllegalArgumentException {
        if (CrossListCapToCheck < 0) {
            throw new IllegalArgumentException("Cross list-cap must be nonnegative.");
        }
    }

    private void checkCRN(int CRN1) throws IllegalArgumentException {
        if (CRN1 < 0) {
            throw new IllegalArgumentException("Section number must be nonnegative.");
        }
    }
    //Unlinked sections and sections that end with 1 are lectures
    public boolean isLecture() {
        return LINK.equals("None") || LINK.endsWith("1");
    }

}
