package edu.odu.cs.cs350;

import java.util.List;

/**
 * TotalCapEnroll calculates and formats the total cap enrollment for courses
 * by aggregating the overall caps of their constituent offerings and sections.
 */
public class TotalCapEnroll {

    private TotalCapEnroll() {};

    /**
     * Returns the total cap enrollment for a given course.
     * 
     * @param course to evaluate
     * @return the total cap enrollment, or 0 if course is null
     */
    public static int calculateTotalCap(Course course) {
        if (course == null || course.getCourseName() == null || course.getCourseName().isBlank() || 
            course.getOfferings() == null || course.getOfferings().isEmpty()) {
            throw new IllegalArgumentException("Course is invalid or missing proper content for calculations.");
        }
        return course.getTotalCap();
    }


    
}
