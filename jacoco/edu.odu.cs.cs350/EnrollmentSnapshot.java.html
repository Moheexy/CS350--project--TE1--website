package edu.odu.cs.cs350;

import java.time.LocalDate;
import java.util.ArrayList;
import java.util.List;

/**
 * 
 * EnrollmentSnapshot; a capture of Course registration data, taken from a particular snapshot.
 */
public class EnrollmentSnapshot {
    /** Date of snapshot. */
    private final LocalDate snapShotDate;
    /** List of course captured in this snapshot. */
    private final ArrayList<Course> courseList;

    /**
     * Constructor for a snapshot.
     * 
     * @param courses listed within the snapshot
     * @param date of the snapshot
     */
    EnrollmentSnapshot(List<Course> courses, LocalDate date)
    {
        this.courseList = new ArrayList<>(courses);
        this.snapShotDate = date;

    }

    /**
     * @return the date of the snapshot
     */
    public LocalDate getSnapShotDateValue() {
        return snapShotDate;
    }
    /**
     * @return the courses listed within the snapshot
     */
    public ArrayList<Course> getCourseList() {
        return courseList;
    }
    
}
