package edu.odu.cs.cs350;

import java.io.FileOutputStream;
import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.Map;
import java.util.function.Function;
import java.util.stream.Collectors;

import org.apache.poi.ss.usermodel.Row;
import org.apache.poi.ss.usermodel.Sheet;
import org.apache.poi.ss.usermodel.Workbook;
import org.apache.poi.xddf.usermodel.chart.AxisPosition;
import org.apache.poi.xddf.usermodel.chart.ChartTypes;
import org.apache.poi.xddf.usermodel.chart.XDDFScatterChartData;
import org.apache.poi.xddf.usermodel.chart.XDDFValueAxis;
import org.apache.poi.xssf.usermodel.XSSFChart;
import org.apache.poi.xssf.usermodel.XSSFClientAnchor;
import org.apache.poi.xssf.usermodel.XSSFDrawing;
import org.apache.poi.xssf.usermodel.XSSFSheet;
import org.apache.poi.xssf.usermodel.XSSFWorkbook;

import edu.odu.cs.cs350.Course.EnrollmentPoint;

/**
 * 
 * Our Detailed Projection Report; given data and creates report file.
 */
public class DetailedReport {

    /** All the semesters we are using for our projection. */
    private final ArrayList<Semester> allSemesters;
    /** The file location where we want our report to be located. */
    private final Path reportLocation;
    /** Shortcut variable for allSemesters[last]. */
    private final Semester lastSemester;

    /**
     * The constructor for our DetailedReport.
     * @param argumentSemesters the semesters we project with
     * @param argumentLocation the location we write the report
     */
    public DetailedReport(ArrayList<Semester> argumentSemesters, Path argumentLocation) {
        allSemesters = argumentSemesters;
        reportLocation = argumentLocation;
        lastSemester = allSemesters.get(allSemesters.size()-1);
    }

    /**
     * Creates our DPR filename.
     * @return the name of our DPR file, based on name of last (current) semester
     */
    public String createDPRFileName()
    {
        String fNameWithSpaces = "Detailed Projection Report-" + lastSemester.getSemesterTitle();
        return fNameWithSpaces.replace(' ','-');
    }

    /** 
     * Writes the actual DPR, in all its glory!
     * 
     * TODO: Call projection in a sane manner as oppsed to what we are currently doing.
     * 
     * @param projectionsList the projections we have already made
    */
    public void writeReport(List<SummaryReport.CourseProjection> projectionsList) throws IOException
    {
        //Collect these into map for easier earch
        Map<String, SummaryReport.CourseProjection> projectionsMap = projectionsList.stream()
        .collect(Collectors.toMap(SummaryReport.CourseProjection::courseName, Function.identity() ));

        //Create directory,decide filename
        Files.createDirectories(reportLocation);
        Path outputSS = reportLocation.resolve(createDPRFileName()+".xlsx");

        //Create workbook
		Workbook DetailedProjectionReportWorkbook = new XSSFWorkbook();

        //Get list of courses from last snapshot
        //Sort by coursename
        var snapshots = lastSemester.getEnrollmentSnapshots();
        var courseList = snapshots.get(snapshots.size()-1).getCourseList();
        courseList.sort(Comparator.comparing(Course::getCourseName));


        for(int i=0; i< courseList.size(); i++)
        {
            Course course = courseList.get(i);

            Sheet courseSheet = DetailedProjectionReportWorkbook.createSheet(course.getCourseName());
            

            Row Header = courseSheet.createRow(0);
            int columnGuide = -1;
            for( Semester semester : allSemesters)
            {
                var enrollmentPoints = Course.getPoints(semester, course.getCourseName());
                
                if(!semester.equals(lastSemester))
                {(Header.createCell(++columnGuide)).setCellValue("d Historical");}
                else {(Header.createCell(++columnGuide)).setCellValue("d Current");}
                (Header.createCell(++columnGuide)).setCellValue(semester.getSemesterTitle());


                createRow(courseSheet, columnGuide, enrollmentPoints);

                //Display projection this course
                if(semester.equals(lastSemester))
                {

                // Note: the explicit return of columnGuide is not required;  this could be handled implicitly by just relying
                // on the pointer being updated, however it is included for clarity's sake.
                columnGuide = configureProjectionCells(projectionsMap, course, courseSheet, Header, columnGuide,
                        enrollmentPoints);
                }

            }
            //Size Column
            for(int columns = 0 ; columns < columnGuide; columns++ )
            {
                courseSheet.autoSizeColumn(columns);
            }
            createEnrollmentChart((XSSFSheet) courseSheet, columnGuide + 3);
        }

        
        //Write
        
        try (FileOutputStream fileOutputStream =
                new FileOutputStream(outputSS.toFile())) {

            DetailedProjectionReportWorkbook.write(fileOutputStream);
        }

        DetailedProjectionReportWorkbook.close();
    }

    /**
     * Configures our projection related cells.
     * 
     * @param projectionsMap mapper from course to projection of course
     * @param course the course we are projecting
     * @param courseSheet the sheet of the course
     * @param Header the header row, which we are appending to
     * @param columnGuide last index of columns we are interested in writing to
     * @param enrollmentPoints the list of enrollment points
     * @return columnGuide, set to its greatest value
     */
    private int configureProjectionCells(Map<String, SummaryReport.CourseProjection> projectionsMap, Course course,
            Sheet courseSheet, Row Header, int columnGuide, ArrayList<EnrollmentPoint> enrollmentPoints) {
        //Projected columns
        //Last enrollment point of current & 
        //Projection at d=1
        (Header.createCell(++columnGuide)).setCellValue("d Projected");
        (Header.createCell(++columnGuide)).setCellValue("Projected");

        Row lastSnapshotRow = courseSheet.getRow(1);
        var lastCurrentSnapshot = enrollmentPoints.get(enrollmentPoints.size()-1);
        lastSnapshotRow.createCell(columnGuide-1).setCellValue(lastCurrentSnapshot.progress());
        lastSnapshotRow.createCell(columnGuide).setCellValue(lastCurrentSnapshot.enrollment());

        //Find THIS course
        SummaryReport.CourseProjection thisCourseProjection = projectionsMap.get(course.getCourseName());
        Row projectionProperRow = courseSheet.getRow(2);
        //by definition, we projected until the addDeadline, d=1
        projectionProperRow.createCell(columnGuide-1).setCellValue(1);
        
        //Realistically, this does not occur unless our input data was unusuably bad
        setProjectionCellProper(columnGuide, thisCourseProjection, projectionProperRow);
        return columnGuide;
    }

    /**
     * Set the value of our projection cell (N/A or the actual projection value).
     * @param columnGuide the column we are writing projection data into
     * @param thisCourseProjection the course projection entity
     * @param projectionProperRow the row we are concerned with
     */
    private void setProjectionCellProper(int columnGuide, SummaryReport.CourseProjection thisCourseProjection,
            Row projectionProperRow) {
        var projection_cell = projectionProperRow.createCell(columnGuide);
        if(thisCourseProjection == null ||
            thisCourseProjection.projectedEnrollment().isEmpty() )
        {
            projection_cell.setCellValue("N/A");
        }
        else
        {
            int knownProjection = thisCourseProjection.projectedEnrollment().getAsInt();
            projection_cell.setCellValue(knownProjection);
        }
    }

    /**
     * Create the component of the row containing snapshot values.
     * 
     * @param courseSheet our detailed projection report sheet
     * @param enrollmentPoints list of progress-enrollemnt points
     * @param columnGuide last index of columns we are interested in writing to
     */
    private void createRow(Sheet courseSheet, int columnGuide, ArrayList<EnrollmentPoint> enrollmentPoints) {
        for(int j =0; j<enrollmentPoints.size(); j++)
        {

            if(courseSheet.getRow(j+1)==null)
            {
                courseSheet.createRow(j+1);
            }

            Row datePointRow = courseSheet.getRow(j+1);
            datePointRow.createCell(columnGuide-1).setCellValue(enrollmentPoints.get(j).progress());
            datePointRow.createCell(columnGuide).setCellValue(enrollmentPoints.get(j).enrollment());
            
        }
    }
    /**
     * Creates a graph.
     * @param sheet the course sheet where the chart is made
     */
    public void createEnrollmentChart(XSSFSheet sheet, int lastColumn) {
    XSSFDrawing drawing = sheet.createDrawingPatriarch();

    // Put the chart to the right of columns
    XSSFClientAnchor anchor = new XSSFClientAnchor(0, 0, 0, 0, lastColumn, 7, lastColumn + 8, 18);
    XSSFChart chart = drawing.createChart(anchor);
    chart.setTitleText("Enrollment Graph");

    XDDFValueAxis xAxis = chart.createValueAxis(AxisPosition.BOTTOM);
    xAxis.setTitle("Enrollment Time Period");

    XDDFValueAxis yAxis = chart.createValueAxis(AxisPosition.LEFT);
    yAxis.setTitle("Enrollment");

    XDDFScatterChartData data = (XDDFScatterChartData) chart.createData(ChartTypes.SCATTER, xAxis, yAxis);

    chart.plot(data);
}

}
