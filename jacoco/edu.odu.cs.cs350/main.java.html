package edu.odu.cs.cs350;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.InvalidPathException;
import java.nio.file.Path;
import java.nio.file.Paths;
import java.time.LocalDate;
import java.time.format.DateTimeParseException;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.TreeSet;

/**
 * 
 * Main class of our program.
 */
public final class main {

    private main(){}

    /**
     * Main method.
     * 
     * @param mainArguments CLI arguments that should be of the form:
     *  args[0-N] = 1-N historical semester directories
     *  args[N+1] = 1 current semester directory
     *  args[N+2] = file system path where the detailed projection report will be written
     *  args[N+3] = a date (YYYY-MM-DD) indicating last day of the curretn semester on which enrollment data should be considered
     */
    public static void main(String[] mainArguments) {

        if(mainArguments.length < 4)
        {
            System.out.print(
            """
            Usage: java -jar EnrollmentProjection.jar [OPTIONS] <SEMESTER_DIRS>...<SEMESTER_TO_PROJECT_DIR> <DPR_TARGET_DIR> <CUTOFF_DATE>
            """);
            return;
        }

        conductProjectionAndDisplay(mainArguments);
    }

   
    private static void conductProjectionAndDisplay(String[] mainArguments) {
        // Newline to separate arguments from program output
        System.out.println();

        // Parse cutoff date
        LocalDate cutoffDate =
            parseCutOffDate(mainArguments[mainArguments.length - 1]);

        // Get DPR output directory
        Path detailedProjectReportOutputPath =
            parseDPROutputLocation(mainArguments[mainArguments.length - 2]);

        if (detailedProjectReportOutputPath == null) return;

        // Parse semester directories
        ArrayList<Path> semesterDirectories =
            parsePotentialSemesterDirectories(
                Arrays.copyOfRange(
                    mainArguments, 0, mainArguments.length - 2
                )
            );

        // Check for at least two semester directories
        if (semesterDirectories.size() < 2) return;

        // Remove duplicate directories
        semesterDirectories = removeDuplicateDirectories(semesterDirectories);

        if (semesterDirectories.size() < 2) {
            System.err.println("Too few unique semester directories entered.");
            return;
        }

        // Create DPR output directory if it doesn't exist
        try {
            Files.createDirectories(detailedProjectReportOutputPath);
        } catch (IOException | SecurityException e) {
            System.err.println(
                "Unable to create DPR output directory: " + e.getMessage()
            );
            return;
        }

        // Read semester data and generate summary report
        try {
            ArrayList<Semester> allSemesters =
                ProjectionReader.readSemesterDirectory(
                    semesterDirectories, cutoffDate
                );

            checkCurrentSemesterPresent(semesterDirectories, allSemesters);

            SummaryReport sumReport = new SummaryReport(allSemesters);
            System.out.println(sumReport.printMenu());

            DetailedReport detailedProjectionReport = new DetailedReport(allSemesters,detailedProjectReportOutputPath);
            detailedProjectionReport.writeReport(sumReport.getProjectedCourses());
        }
        catch (IOException | IllegalArgumentException e)
        {
            System.err.println("Unable to calculate enrollment projection: " + e.getMessage());
        }
    }

    /**
     * Stop our program if the last user supplied directory (the one they want a projection for) was invalid.
     * 
     * @param semesterDirectories the user supplied directories
     * @param allSemesters the semesters list our program came up with
     * @throws IllegalArgumentException if the last semester and last directory don't match (semestercode)
     */
    public static void checkCurrentSemesterPresent(ArrayList<Path> semesterDirectories, ArrayList<Semester> allSemesters) {
        //Check if we rejected current
        if(!semesterDirectories.get(semesterDirectories.size()-1).endsWith(allSemesters.get(allSemesters.size()-1).getSemesterCode()))
        {
            String badCurrent = "Current semester path was rejected. Please ensure the path and files within are properly formed.";
            System.out.println(badCurrent);
            throw new IllegalArgumentException(badCurrent);
        }
    }

    /**
     * Removes duplicated semesterDirectories, starting backwards from current (last index).
     * 
     * @param semesterDirectories
     * @return the list of unique semesterDirectories
     */
    public static ArrayList<Path> removeDuplicateDirectories(ArrayList<Path> semesterDirectories) {
        TreeSet<Path> knownDirPaths = new TreeSet<>();
        for(int pathIndex = semesterDirectories.size()-1 ; pathIndex >-1; pathIndex--)
        {
            Path pathToCheck = semesterDirectories.get(pathIndex);
            if(knownDirPaths.contains(pathToCheck))
            {
               System.out.println("Duplcate path entered: \"" + pathToCheck.toString() + "\", omitting directory.");
               semesterDirectories.remove(pathIndex);
            }
            knownDirPaths.add(pathToCheck);
        }

        return semesterDirectories;
    }

    /**
     * Basic parsing of potential Semester Directories, determines if they are valid paths to directories.
     * 
     * @param potentialSemesterDirectories the directories to consider
     * @return a collection of valid 0-(N+1) directories
     */
    public static ArrayList<Path> parsePotentialSemesterDirectories(String[] potentialSemesterDirectories) {

        ArrayList<Path> semesterDirectories = new ArrayList<>();

        for(int argumentIndex = 0; argumentIndex < potentialSemesterDirectories.length; argumentIndex++)
        {
            try {

                Path potentialDirectoryPath = Paths.get(potentialSemesterDirectories[argumentIndex]);


                if(Files.isDirectory(potentialDirectoryPath))
                {

                    semesterDirectories.add(potentialDirectoryPath);
                }

                
            } 
            catch (InvalidPathException e) {
                System.out.print("Invalid file path entered, halting program. Confirm a path exists for argument " + (argumentIndex+1)  + ".");
            }
        }

        if(semesterDirectories.size() < 2)
        {
            System.out.print("Too few semester directories entered. Please ensure you have entered at least two semester directories.");
        }

        return semesterDirectories;
    }


    /**
     * Basic parsing of a potential file location for the Detailed Projection Report.
     * 
     * Since garbage command line input is common, a bad DPR is indicated through null
     * NOT an exception; a rubbish path is a bad case, but not exceptional.
     * 
     * @param potentialDPRLocationPath a potential file location for the DPR
     * @return the detailed projection report, or null
     */
    public static Path parseDPROutputLocation(String potentialDPRLocationPath) {
        Path detailedProjectionReportPath = null;
        try {
            detailedProjectionReportPath = Paths.get(potentialDPRLocationPath);
        } catch (InvalidPathException e) {
            System.out.print("Invalid location for Detailed Projection Report output requested. Review the path you entered.");

        }
        
        return detailedProjectionReportPath;
    }

    /**
     * Basic parsing on a potential cutoff date as a string.
     * 
     * @param potentialCutoffDate the cutoff string to consider
     * @return a LocalDate, either taken from the string or assigned as a default
     */    
    public static LocalDate parseCutOffDate(String potentialCutoffDate) {
        LocalDate cutOffDate;
        try
        {

        cutOffDate = LocalDate.parse(potentialCutoffDate);
        }

        catch (DateTimeParseException e)
        {
            System.out.print(
            """
            Invalid cutoff date (YYYY-MM-DD) provided for the current semester. Setting to registration add deadline (see dates.txt in current semester directory).
            """);
            //Set a default value for cutOffDate, to be reduced later
            //This gets assigned to add-deadline when we form a semester in ProjectionReader
            cutOffDate = LocalDate.of(999999999,10,10);
        }

        return cutOffDate;
    }
}
