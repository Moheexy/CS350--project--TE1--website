package edu.odu.cs.cs350;

import java.io.File;
import java.io.FileNotFoundException;
import java.io.IOException;
import java.nio.file.DirectoryStream;
import java.nio.file.Files;
import java.nio.file.Path;
import java.time.LocalDate;
import java.time.format.DateTimeParseException;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.HashMap;
import java.util.List;
import java.util.Map.Entry;
import java.util.Scanner;
import java.util.Set;

import com.opencsv.CSVReader;
import com.opencsv.CSVReaderBuilder;
import com.opencsv.exceptions.CsvException;

/**
 * 
 * ProjectionReader; Reads semester directories and creates the proper nesting of Semesters - Snapshots - Courses - Offerings - Sections.
 * 
 */
public class ProjectionReader {

    private ProjectionReader() {
    }

    /**Headers we want to parse. */
    private static final Set<String> desiredHeaders = Set.of(
            "CRN", "SUBJ", "CRSE", "XLST CAP", "ENR", "LINK", "XLST GROUP", "OVERALL CAP", "OVERALL ENR");

    /**
     * Builds the core data structure of this program.
     * 
     * @param semesterDirectories directories containing semester data
     * @param cutoffDate          for the last semester
     * @return a list of semesters, each populated with a nesting of Snapshots,
     *         Courses, Offerings, and Sections
     * @throws IOException in case of a missing dates.txt file and other I/O related
     *                     processing
     */
    public static ArrayList<Semester> readSemesterDirectory(ArrayList<Path> semesterDirectories, LocalDate cutoffDate)
            throws IOException {
        ArrayList<Semester> allSemesters = new ArrayList<>();

        // For all semesters
        var semesterDirectoriesIterator = semesterDirectories.iterator();

        //TODO: Notify user when they enter a nonexistent directory
        while (semesterDirectoriesIterator.hasNext()) {
            // The semester and its status as/not as the last semester
            Path semesterDirectory = semesterDirectoriesIterator.next();
            boolean isLastSemester = !semesterDirectoriesIterator.hasNext();
            LocalDate semesterCutoff = cutoffDate;

            // Semester information containers
            ArrayList<EnrollmentSnapshot> snapshots = new ArrayList<>();
            String semesterCode = semesterDirectory.getFileName().toString();
            LocalDate dateOfPreregistration;
            LocalDate addDeadlineDate;

            // Get date borders
            File preRegistrationAddDeadlineDatesFile = new File(
                    (semesterDirectory.toString() + File.separator + "dates.txt"));

            ArrayList<LocalDate> dateBorders = dateBorders(preRegistrationAddDeadlineDatesFile, semesterCode);
            dateOfPreregistration = dateBorders.get(0);
            addDeadlineDate = dateBorders.get(1);

            // User provided cutoff is before preregistration of lastsemester, throw
            if (isLastSemester && cutoffDate.isBefore(dateOfPreregistration)) {
                throw new IllegalArgumentException(
                        "Please provide a cutoff date on or after the first date of pre-registration for the semester under consideration");
            }

            semesterCutoff = chooseCutoff(isLastSemester, semesterCutoff, addDeadlineDate);

            try (DirectoryStream<Path> SemesterSnapshots = Files.newDirectoryStream(semesterDirectory, "*.csv")) {

                // added debugging output to help user (me) understand which directory it not
                // valid
                // System.err.println("Reading semester directory: " + semesterCode);

                readSnapshots(allSemesters, isLastSemester, semesterCutoff, snapshots, semesterCode,
                        dateOfPreregistration, addDeadlineDate, SemesterSnapshots);

                // debugging line for finding bad snapshots to figure out why test is failing
                // System.err.println("Finished reading: " + semesterCode + " | Valid snapshots
                // found: " + snapshots.size());
            }

        }
        // DirectoryStream fails
        // catch (IOException e) { System.err.println(String.format("Error reading
        // directory %s: " + e.getMessage(), semesterCode));}

        if (allSemesters.size() < 2) {

            throw new IllegalArgumentException(
                    "Please provide at least two properly named semester directories with snapshots taken between pre-registration and the add deadline.");

        }

        return allSemesters;

    }

    private static void readSnapshots(ArrayList<Semester> allSemesters, boolean isLastSemester,
            LocalDate semesterCutoff, ArrayList<EnrollmentSnapshot> snapshots, String semesterCode,
            LocalDate dateOfPreregistration, LocalDate addDeadlineDate, DirectoryStream<Path> SemesterSnapshots)
            throws IOException {
        /* These go inside a Snapshot (Courses, Offering, Sections) */

        // sort list of snapshots by date, so that we can process them in order
        var sortedSnapshots = sortSnapshotsEagerly(SemesterSnapshots);

        // This flag belongs to THIS loop, so only this method may change it:
        // a kept post-deadline snapshot uses up the one allowed slot.
        boolean includedPostDeadlineSnapshot = false;

        // For all snapshots
        for (Path snapshot : sortedSnapshots) {

            // debugging line for finding bad snapshots to figure out why test is failing
            // when the files should be accepted
            // System.err.println("Checking snapshot: " + snapshot.getFileName());

            includedPostDeadlineSnapshot = processSnapshot(isLastSemester, semesterCutoff, snapshots,
                    semesterCode, dateOfPreregistration, addDeadlineDate, includedPostDeadlineSnapshot, snapshot);
        }
        
        // TODO: Inform user if they gave us a bad semester (i.e too few snapshots)
        // TODO: need unique exception if this was the last semester
        try {
            // Validate the number of snapshots in the semester then add it
            validateSnapshotCount(snapshots, semesterCode);
            allSemesters.add(new Semester(
                    snapshots,
                    dateOfPreregistration,
                    semesterCutoff,
                    semesterCode,
                    addDeadlineDate));
        } catch (Exception e) {
            System.err.println(String.format(
                    "Unable to use semester directory %s: %s", semesterCode, e.getMessage()));
        }
    }

    private static boolean processSnapshot(boolean isLastSemester, LocalDate semesterCutoff,
            ArrayList<EnrollmentSnapshot> snapshots, String semesterCode, LocalDate dateOfPreregistration,
            LocalDate addDeadlineDate, boolean includedPostDeadlineSnapshot, Path snapshot) throws IOException {
        // Skip subdirectories
        if (!Files.isRegularFile(snapshot))
            return includedPostDeadlineSnapshot;

        // check for bad file names, skip if not a valid date
        LocalDate snapshotDate;
        try {
            snapshotDate = getSnapshotDateValue(snapshot);
        } catch (DateTimeParseException e) {
            System.err.println(String.format(
                    "Error reading csv file %s in semester directory %s: " + e.getMessage(),
                    snapshot.getFileName().toString(), semesterCode));
            return includedPostDeadlineSnapshot;
        }

        // Ignore snapshot if outside bounds
        if (checkSkipSection(isLastSemester, snapshotDate, semesterCutoff,
                dateOfPreregistration, addDeadlineDate, includedPostDeadlineSnapshot)) {

            return includedPostDeadlineSnapshot;
        }

        // the flag lives here.
        if (!isLastSemester && snapshotDate.isAfter(addDeadlineDate)) {
            includedPostDeadlineSnapshot = true;
        }

        ArrayList<Course> snapshotCourses = new ArrayList<>();

        // Every course offering must match coursename
        HashMap<String, ArrayList<Offering>> snapshotOfferingGroups = new HashMap<>();

        // Some sections do not have a particular offering group they map to. In this
        // case, we must store theses offerings and sections discretely.
        ArrayList<Offering> soleSectionOfferings = new ArrayList<>();

        // Every offering section must match crossListGroup
        HashMap<String, ArrayList<Section>> snapshotSectionGroups = new HashMap<>();

        try (CSVReader snapshotReader = new CSVReaderBuilder(Files.newBufferedReader(snapshot))
                .withSkipLines(0)
                .build()) {

            ReadCSVRows(snapshots, semesterCode, snapshot, snapshotDate, snapshotCourses,
                    snapshotOfferingGroups, soleSectionOfferings, snapshotSectionGroups, snapshotReader);

            // debugging line for finding bad snapshots to figure out why test is failing
            // when the files should be accepted
            // System.err.println("Added snapshot: " + snapshot.getFileName());
        }
        return includedPostDeadlineSnapshot;
    }

    private static void ReadCSVRows(ArrayList<EnrollmentSnapshot> snapshots, String semesterCode, Path snapshot,
            LocalDate snapshotDate, ArrayList<Course> snapshotCourses,
            HashMap<String, ArrayList<Offering>> snapshotOfferingGroups, ArrayList<Offering> soleSectionOfferings,
            HashMap<String, ArrayList<Section>> snapshotSectionGroups, CSVReader snapshotReader) {
        try {

            List<String[]> allFields = snapshotReader.readAll();
            List<String[]> rows = allFields.subList(1, allFields.size() - 1);

            HashMap<String, Integer> headersToColumns = getDesiredHeaderIndices(allFields.get(0));

            // For all sections
            for (String[] fields : rows) {

                Section rowSection = createSection(fields, headersToColumns);

                // We do not consider labs and recitations
                if (!rowSection.isLecture())
                    continue;

                divideSectionsIntoOfferingGroups(soleSectionOfferings, snapshotSectionGroups,
                        rowSection);

                // System.out.println(rowSection);
            }

            formCoursesFromAllOfferings(snapshotCourses, snapshotOfferingGroups, soleSectionOfferings,
                    snapshotSectionGroups);

            // Create a snapshot from our list of courses taken in this snapshot
            snapshots.add(new EnrollmentSnapshot(snapshotCourses, snapshotDate));
        }
        // getDesiredHeaderIndices fails
        catch (IOException | CsvException e) {
            System.err.println(String.format(
                    "Error reading csv file %s in semester directory %s: " + e.getMessage(),
                    snapshot.getFileName().toString(), semesterCode));

        }
    }

    /**
     * Decides whether a snapshot file should be skipped.
     *
     * Pure decision function: it never changes state; the caller owns the
     * includedPostDeadlineSnapshot flag and updates it after keeping a
     * post-deadline snapshot.
     *
     * @param isLastSemester               true for the semester being projected
     * @param snapshotDate                 the parsed date of the snapshot file
     * @param semesterCutoff               the cutoff in effect for this semester
     * @param dateOfPreregistration        the semester's pre-registration date
     * @param addDeadlineDate              the semester's add deadline
     * @param includedPostDeadlineSnapshot whether this historical semester has
     *                                     already kept its one allowed
     *                                     post-deadline snapshot
     * @return true if the snapshot must be skipped
     */
    private static boolean checkSkipSection(boolean isLastSemester, LocalDate snapshotDate,
            LocalDate semesterCutoff, LocalDate dateOfPreregistration, LocalDate addDeadlineDate,
            boolean includedPostDeadlineSnapshot) {

        // The user-supplied cutoff applies only to the semester being projected.
        // Historical semesters are cut off at their own add deadline (see
        // chooseCutoff); their single allowed post-deadline snapshot is
        // handled below, so it must NOT be skipped by this bounds check.
        if (isLastSemester && snapshotDate.isAfter(semesterCutoff)) {
            return true;
        }

        if (snapshotDate.isBefore(dateOfPreregistration)) {
            return true;
        }

        /**
         * Historical semesters keep at most ONE post-deadline snapshot
         * (the first, since we're sorted). A point past 1.0 is required to
         * interpolate h(1.0), keeping all post-deadline files would skew
         * the interpolation so only the first one is kept
         */
        return !isLastSemester
                && snapshotDate.isAfter(addDeadlineDate)
                && includedPostDeadlineSnapshot;// reads flag instead of writing it
    }

    private static ArrayList<Path> sortSnapshotsEagerly(DirectoryStream<Path> SemesterSnapshots) {
        ArrayList<Path> sortedSnapshots = new ArrayList<>();
        for (Path snapshot : SemesterSnapshots) {
            sortedSnapshots.add(snapshot);
        }
        sortedSnapshots.sort(Comparator.naturalOrder());
        return sortedSnapshots;
    }

    /**
     * Verifies that a semester contains enough enrollment snapshots to be used
     * for projection calculations.
     *
     * @param snapshots    the snapshots found for the semester
     * @param semesterCode the semester being checked
     * @throws IOException if fewer than two valid snapshots are available
     */
    private static void validateSnapshotCount(
            ArrayList<EnrollmentSnapshot> snapshots,
            String semesterCode) throws IOException {

        if (snapshots.size() < 2) {
            throw new IOException(
                    String.format("Insufficient snapshots in %s.", semesterCode));
        }
    }

    /**
     * Forms a section.
     * 
     * IMPORTANT: If there is no provided cap; this means the course is sole
     * offering,
     * thus we assign cap to XLSTCAP, since this is the only cap of enrollment for
     * the implicit section - offering pair.
     * 
     * @param fields           the csv fields we are drawing our section from
     * @param headersToColumns mapping from header labels to column rows
     * @return a fully formed section
     * @throws NumberFormatException when we fail parse a field we expect to be
     *                               numeric
     */
    public static Section createSection(String[] fields, HashMap<String, Integer> headersToColumns)
            throws NumberFormatException {

        // Define what we want from the row and create the section
        int CRN = Integer.parseInt(fields[headersToColumns.get("CRN")]);
        String subject = fields[headersToColumns.get("SUBJ")];
        String CourseNumber = fields[headersToColumns.get("CRSE")];
        int CrossListCap = Integer.parseInt(fields[headersToColumns.get("XLST CAP")]);
        int ENR = Integer.parseInt(fields[headersToColumns.get("ENR")]);
        String LINK = fields[headersToColumns.get("LINK")];
        String crossListGroup = fields[headersToColumns.get("XLST GROUP")];
        int overallEnrollmentCap = (fields[headersToColumns.get("OVERALL CAP")].trim().isEmpty())
                ? CrossListCap
                : Integer.parseInt(fields[headersToColumns.get("OVERALL CAP")]);
        int overallEnrollment = Integer.parseInt(fields[headersToColumns.get("OVERALL ENR")]);
        Section rowSection = new Section(CRN, subject, CourseNumber, CrossListCap, ENR, LINK, crossListGroup,
                overallEnrollmentCap, overallEnrollment);
        return rowSection;
    }

    /**
     * Logic for choosing cutoff date for semesters.
     * Core Implementation of User Story [ Command Line Input : Cutoff Date ]
     * 
     * @param isLastSemester  if the dates belong to the last semester, the one we
     *                        are projecting
     * @param semesterCutoff  the semester cutoff actually applied to the semester,
     *                        initialized from CLI input
     * @param addDeadlineDate the add deadline, initialized from dates.txt
     * @return the proper cutoff
     */
    public static LocalDate chooseCutoff(boolean isLastSemester, LocalDate semesterCutoff, LocalDate addDeadlineDate) {

        // Cutoff is always determined by dates.txt unless we are evaluating the
        // current(last) semester
        if (!isLastSemester)
            return addDeadlineDate;

        // In which case we will choose the most conservative cutoff
        if (semesterCutoff.compareTo(addDeadlineDate) > 0)
            return addDeadlineDate;

        return semesterCutoff;
    }

    /**
     * Forms courses from the appropriate sole-section and multi-section offerings.
     * 
     * @param snapshotCourses        the list of courses populated
     * @param snapshotOfferingGroups the collection of multi-section offerings,
     *                               mapped to course name
     * @param soleSectionOfferings   the collection of sole-section offerings
     * @param snapshotSectionGroups  the collection of multi-section sections,
     *                               mapped to multi-section offering identifier
     *                               (crossListGroup)
     */
    public static void formCoursesFromAllOfferings(
            ArrayList<Course> snapshotCourses,
            HashMap<String, ArrayList<Offering>> snapshotOfferingGroups,
            ArrayList<Offering> soleSectionOfferings,
            HashMap<String, ArrayList<Section>> snapshotSectionGroups) {

        // Create an offering that has sections with all matching crossListGroups
        // Map this offering to its coursename
        for (ArrayList<Section> crossListGroupSections : snapshotSectionGroups.values()) {
            Offering crossListGroupOffering = new Offering(crossListGroupSections);

            String offeringCourse = crossListGroupOffering.getCourseName();

            if (snapshotOfferingGroups.containsKey(offeringCourse)) {
                snapshotOfferingGroups.get(offeringCourse).add(crossListGroupOffering);
            } else {
                snapshotOfferingGroups.put(offeringCourse, new ArrayList<>(List.of(crossListGroupOffering)));
            }
        }

        // Append sole section offerings as distinct offerings of a particular
        // coursename
        for (Offering soleOffering : soleSectionOfferings) {
            String soleOfferingCourseName = soleOffering.getCourseName();
            if (snapshotOfferingGroups.containsKey(soleOfferingCourseName)) {
                snapshotOfferingGroups.get(soleOfferingCourseName).add(soleOffering);
            } else {
                snapshotOfferingGroups.put(soleOfferingCourseName, new ArrayList<>(List.of(soleOffering)));
            }
        }

        // Create and add a course from our complete list of offerings
        for (Entry<String, ArrayList<Offering>> courseOfferingPair : snapshotOfferingGroups.entrySet()) {
            snapshotCourses.add(new Course(courseOfferingPair.getValue()));
        }
    }

    /**
     * Directs a paricular section into either a map of sections linked to the same
     * offering or a collection of sections who comprise one unique offering.
     * 
     * @param soleSectionOfferings  a collection of sections which comprise singular
     *                              offerings
     * @param snapshotSectionGroups a map of offerings to collections of linked
     *                              sections
     * @param rowSection            the section under consideration
     */
    public static void divideSectionsIntoOfferingGroups(
            ArrayList<Offering> soleSectionOfferings,
            HashMap<String, ArrayList<Section>> snapshotSectionGroups,
            Section rowSection) {

        // If section does not have an crossListGroup we can simply add it to the other
        // solo
        // offerings
        if (rowSection.crossListGroup().equals("None")) {
            Offering soleSectionOffering = new Offering(List.of(rowSection));
            soleSectionOfferings.add(soleSectionOffering);
        }
        // If the section has an crossListGroup, we must map it to that group for
        // organized
        // appending
        else {
            String singleOfferingIdentifier = rowSection.crossListGroup();

            /*
             * Note: CourseNumber (CRSE) CANNOT be included in this as this will prevent
             * proper grouping of
             * 400/500 level mixed courses
             * Even subject is not a reliable component of this, however is is acceptable
             * since
             * all files we will process are guarenteed to
             * only have CS courses.
             */

            if (snapshotSectionGroups.containsKey(singleOfferingIdentifier)) {

                snapshotSectionGroups.get(singleOfferingIdentifier).add(rowSection);
            } else {
                snapshotSectionGroups.put(singleOfferingIdentifier, new ArrayList<>(List.of(rowSection)));
            }
        }
    }

    /**
     * Finds the column numbers for our desired headers.
     * 
     * @param headers the header row to check
     * @return a map of desired headers to column indices
     * @throws IOException if any of the desired headers are absent
     */
    public static HashMap<String, Integer> getDesiredHeaderIndices(String[] headers) throws IOException {

        HashMap<String, Integer> headersToColumns = new HashMap<>();

        for (int headerIndex = 0; headerIndex < headers.length; headerIndex++) {
            String headerFieldValue = headers[headerIndex];
            if (desiredHeaders.contains(headerFieldValue)) {
                headersToColumns.put(headerFieldValue, headerIndex);
            }

        }

        ArrayList<String> missingHeaders = new ArrayList<>();

        for (String desiredHeader : desiredHeaders) {
            if (!headersToColumns.containsKey(desiredHeader)) {
                missingHeaders.add(desiredHeader);
            }
        }

        // This pretty much just exists for Unit Testing
        if (!missingHeaders.isEmpty()) {
            missingHeaders.sort(Comparator.naturalOrder());// hashing means these could be in any order; can't easily
                                                           // test a randomized string so sort

            StringBuilder allMissingHeaders = new StringBuilder();
            for (int missingHeaderIndex = 0; missingHeaderIndex < missingHeaders.size(); missingHeaderIndex++) {
                allMissingHeaders.append(missingHeaders.get(missingHeaderIndex));
                allMissingHeaders.append(", ");
            }
            allMissingHeaders.deleteCharAt(allMissingHeaders.length() - 1);// remove extra space
            allMissingHeaders.deleteCharAt(allMissingHeaders.length() - 1);// remove extra comma

            throw new IOException(String.format("Illegally formed csv file. The field(s) [%s] are missing.",
                    allMissingHeaders.toString()));

        }

        return headersToColumns;
    }

    /**
     * Computes the preRegistration and addDeadline.
     * 
     * @param registrationDatesFile the dates.txt file, which contains our cutoffs
     * @param semesterCode          the semester we are checking
     * @return an arraylist containing the two dates
     * @throws FileNotFoundException
     */
    public static ArrayList<LocalDate> dateBorders(File registrationDatesFile, String semesterCode) throws IOException {
        LocalDate dateOfPreregistration;
        LocalDate addDeadlineDate;

        try (Scanner myReader = new Scanner(registrationDatesFile)) {
            try {
                // before reading date, check that it exists
                if (!myReader.hasNext()) {
                    throw new IOException(String.format("Insufficient dates in %s", semesterCode));
                }

                dateOfPreregistration = LocalDate.parse(myReader.next());

                // before reading date, check that it exists
                if (!myReader.hasNext()) {
                    throw new IOException(String.format("Insufficient dates in %s", semesterCode));
                }

                addDeadlineDate = LocalDate.parse(myReader.next());

            } catch (DateTimeParseException e) {
                // Bad dates.txt strings, throw IOException
                throw new IOException(String.format("Invalid dates.txt file in %s", semesterCode));
            }

        }
        // We can't proceed without dates.txt, so we have to throw
        catch (FileNotFoundException e) {
            throw new IOException(String.format("Missing dates.txt in %s", semesterCode));
        }

        return new ArrayList<>(List.of(dateOfPreregistration, addDeadlineDate));

    }

    private static LocalDate getSnapshotDateValue(Path snapshot) {
        return LocalDate.parse((snapshot.getFileName().toString()).replace(".csv", ""));
    }
