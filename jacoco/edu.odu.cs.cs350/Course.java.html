package edu.odu.cs.cs350;

import java.time.LocalDate;
import java.time.temporal.ChronoUnit;
import java.util.ArrayList;
import java.util.Comparator;
import java.util.List;
import java.util.OptionalDouble;
import java.util.OptionalInt;

/**
 * 
 * A course, which is composed of a course name, offerings and enrollment.
 */
public final class Course {
    /**
     * The name of the course.
     */
    private final String courseName;
    /**
     * List of course offerings.
     */
    private final ArrayList<Offering> offerings;
    /**
     * The total enrollment for this course.
     */
    private final int totalEnrollment;
    /**
     * The total cap of enrollment for this course.
     */
    private final int totalCap;

    /**
     * Course constructor. 
     * 
     * @param offerings the offerings of this course
     * @throws IllegalArgumentException if their is not at least one offering associated with the course
     */
    public Course(List<Offering> offerings) throws IllegalArgumentException
    {
        this.offerings = checkOfferings(offerings);
        this.courseName = offerings.get(0).getCourseName();
        this.totalEnrollment = sumTotalEnrollment();
        this.totalCap = findTotalCap();

    }

    private ArrayList<Offering> checkOfferings(List<Offering> listOfOfferings) throws IllegalArgumentException
    {
        if(listOfOfferings.size()<1)
        {
            throw new IllegalArgumentException("A course must have at least one offering.");
        }

        return new ArrayList<>(listOfOfferings);
    }

    /**
     * @return the course name
     */
    public String getCourseName() {
        return courseName;
    }

    /**
     * 
     * @return the course offerings
     */
    public ArrayList<Offering> getOfferings() {
        return offerings;
    }
    /**
     * 
     * Private method for summing up enrollment over offerings.
     * @return the sum of enrollments over all offerings
     */
    private int sumTotalEnrollment()
    {
        int enrollmentSum = 0;
        for(Offering courseOffering : this.offerings)
        {
            enrollmentSum+= courseOffering.getOverallEnrollmentValue();
        }

        return enrollmentSum;
    }

    /**
     * 
     * Private method for summing up cap over offerings.
     * @return the total cap over all offerings
     */
    private int findTotalCap()
    {
        int totalCapEnrollment = 0;
        for(Offering courseOffering : this.offerings)
        {
            totalCapEnrollment+= courseOffering.getOverallEnrollmentValueCap();
        }

        return totalCapEnrollment;
    }

    /**
     * 
     * @return the total enrollment in this course, derived from enrollment in offerings
     */
    public int findTotalEnrollment() {
        return totalEnrollment;
    }

    /**
     * 
     * @return the total cap for this course, derived from caps in offerings
     */
    public int getTotalCap() {
        return totalCap;
    }

    /**
     * Conducts enrollment projection over a list of semesters.
     * 
     * @param semesters the list of semesters to compute a projection over
     * @return a summary report containing the projection information
     */
    public static SummaryReport projectEnrollments(List<Semester> semesters) {
        if (semesters.size() < 2) {
            throw new IllegalArgumentException(
                "At least one historical semester and one current semester are required.");
        }
        //By CLI contract, the last directory is the current semester.
        Semester currentSemester = semesters.get(semesters.size() - 1);
        //d′ — the most recent snapshot we have for the current semester.
        EnrollmentSnapshot currentSnapshot = latestSnapshot(currentSemester);

        //Where d′ sits on the 0…1 scale (0 = preregistration, 1 = add deadline)
        double currentProgress = progressAt(currentSemester, currentSnapshot.getSnapShotDateValue());
       
        ArrayList<Course> currentCourses = copyAndDropZeroLecture(currentSnapshot);

        ArrayList<SummaryReport.CourseProjection> projections = new ArrayList<>();
        for (Course currentCourse : currentCourses) {
            long projectionTotal = 0;
            int usableSemesters = 0;

            for (int i = 0; i < semesters.size() - 1; i++) {
                Semester historicalSemester = semesters.get(i);
                int j = 1;
                //The whole per-semester computation sits inside the try so that
                //ANY unusable semester (bad ratio, zero-day period, bad dates)
                //is skipped the same way instead of killing the whole report.
                try {
                    //Compute h(d′) (history at the same relative point) and h(1) (history at its add deadline). 
                    // OptionalDouble because the course may not exist in that semester, 
                    // or the needed dates may fall outside the historical snapshots.
                    OptionalDouble historicalAtCurrentProgress = enrollmentAt(
                        historicalSemester, currentCourse.getCourseName(), currentProgress);
                    OptionalDouble historicalAtDeadline = enrollmentAt(
                        historicalSemester, currentCourse.getCourseName(), 1.0);
                    //this historical semester can't help; skip it
                    if (historicalAtCurrentProgress.isEmpty() || historicalAtDeadline.isEmpty()) {
                        continue;
                    }

                    projectionTotal += EnrollmentProjection.projectEnrollment(
                        currentCourse.findTotalEnrollment(),
                        historicalAtCurrentProgress.getAsDouble(),
                        historicalAtDeadline.getAsDouble());
                    usableSemesters++;
                } catch (IllegalArgumentException exception) {
                    System.err.printf(
                        "Skipping projection for %s in %s: %s%n",
                        currentCourse.getCourseName(),
                        historicalSemester.getSemesterTitle(),
                        exception.getMessage());
                    // TODO?    
                    // System.out.println(j + " projections skipped");
                    // j++;
                }
            }

            //Refactored for cleain coding (:
            getProjectedEnrollment(projections, currentCourse, projectionTotal, usableSemesters);
        }

        return new SummaryReport(currentSemester, projections);
    }

    private static void getProjectedEnrollment(ArrayList<SummaryReport.CourseProjection> projections,
            Course currentCourse, long projectionTotal, int usableSemesters) {
        OptionalInt projectedEnrollment = usableSemesters == 0
            ? OptionalInt.empty()
            : OptionalInt.of((int) Math.round((double) projectionTotal / usableSemesters));
        projections.add(new SummaryReport.CourseProjection(
            currentCourse.getCourseName(),
            currentCourse.findTotalEnrollment(),
            projectedEnrollment,
            currentCourse.getTotalCap()));
    }
    

    /**
     * Helper method to copy and drop no capacity lecture sections (should be unneeded).
     * @param currentSnapshot to rereive courses from
     * @return a list of courses
     */
    private static ArrayList<Course> copyAndDropZeroLecture(EnrollmentSnapshot currentSnapshot) {
        //Copy list (so snapshot isn't mutated), drop courses with zero lecture capacity
        ArrayList<Course> currentCourses = new ArrayList<>(currentSnapshot.getCourseList());
        currentCourses.removeIf(course -> course.getTotalCap() == 0);
        currentCourses.sort(Comparator.comparing(Course::getCourseName));
        return currentCourses;
    }

    protected static double progressAt(Semester semester, LocalDate date) {
        long periodDays = ChronoUnit.DAYS.between(
            semester.findPreregistrationDate(), semester.getAddDeadline());
        if (periodDays <= 0) {
            throw new IllegalArgumentException(
                "The enrollment period must have a positive number of days.");
        }
        return (double) ChronoUnit.DAYS.between(semester.findPreregistrationDate(), date)
            / periodDays;
    }

    //Interpolation
    //Build (progress, enrollment) pairs from every snapshot where the course exists.
    private static OptionalDouble enrollmentAt(
        Semester semester,
        String courseName,
        double targetProgress) {

        ArrayList<EnrollmentPoint> points = getPoints(semester, courseName);
        
        //Case exact match, no interpolation
        EnrollmentPoint lower = null;
        EnrollmentPoint upper = null;
        for (EnrollmentPoint point : points) {
            if (Double.compare(point.progress(), targetProgress) == 0) {
                return OptionalDouble.of(point.enrollment());
            }
            //Points are sorted by progress, so the last point below the
            //target is the closest one below, and the first point above
            //it is the closest one above; no extra comparisons needed.
            if (point.progress() < targetProgress) {
                lower = point;
            } else if (upper == null) {
                upper = point;
            }
        }
        
        return finalizeInterpolation(targetProgress, lower, upper);
    }

    private static OptionalDouble finalizeInterpolation(double targetProgress, EnrollmentPoint lower, EnrollmentPoint upper) {
        //Target outside the historical data (before the first or after the last snapshot) cannot interpolate
        if (lower == null || upper == null) {
            return OptionalDouble.empty();
        }
        // h(d) = h(dₐ) + (d−dₐ)/(d_b−dₐ) · (h(d_b) − h(dₐ))
        double fraction = (targetProgress - lower.progress())
            / (upper.progress() - lower.progress());
        return OptionalDouble.of(
            lower.enrollment() + fraction * (upper.enrollment() - lower.enrollment()));
    }

    protected static ArrayList<EnrollmentPoint> getPoints(Semester semester, String courseName) {
        ArrayList<EnrollmentPoint> points = new ArrayList<>();
        for (EnrollmentSnapshot snapshot : semester.getEnrollmentSnapshots()) {
            Course course = findCourse(snapshot, courseName);
            if (course != null) {
                points.add(new EnrollmentPoint(
                    progressAt(semester, snapshot.getSnapShotDateValue()),
                    course.findTotalEnrollment()));
            }
        }
        //Chronological order on the 0...1 scale.
        points.sort(Comparator.comparingDouble(EnrollmentPoint::progress));
        return points;
    }
    //A semester with no snapshots 
    //The Semester constructor guarantees snapshots are non-empty and sorted
    //chronologically, so the last element is the latest snapshot.
    private static EnrollmentSnapshot latestSnapshot(Semester semester) {
        List<EnrollmentSnapshot> snapshots = semester.getEnrollmentSnapshots();
        return snapshots.get(snapshots.size() - 1);
    }

    private static Course findCourse(EnrollmentSnapshot snapshot, String courseName) {
        return snapshot.getCourseList().stream()
            .filter(course -> course.getCourseName().equals(courseName))
            .findFirst()
            .orElse(null);
    }

    protected record EnrollmentPoint(double progress, int enrollment) {
    }

    /**
     * Calculates the enrollment expected on the add-deadline date of the current
     * semester, based on how a course's enrollment grew between the equivalent
     * snapshot date and its add deadline in a past semester.
     */
    public static final class EnrollmentProjection {

        /** Static utility class. */
        private EnrollmentProjection() {}

        /**
         * Estimate the enrollment of a course offering on the current semester's
         * add deadline, assuming enrollment grows proportionally to history:
         * c(1) / c(d') = h(1) / h(d'), solved for c(1).
         *
         * @param currentSnapshotEnrollment students enrolled in the offering as of
         *        the latest snapshot date of the current semester; nonnegative
         * @param historicalSnapshotEnrollment students enrolled in the course at
         *        the equivalent snapshot date in a past semester; nonnegative
         * @param historicalDeadlineEnrollment students enrolled in the course on
         *        the add deadline in a past semester; nonnegative
         * @return the projected enrollment, rounded to the nearest whole student
         * @throws IllegalArgumentException if any argument is negative, or if the
         *         historical enrollment grew from zero to a positive value
         */
        public static int projectEnrollment(int currentSnapshotEnrollment, int historicalSnapshotEnrollment, int historicalDeadlineEnrollment) {
            //Int overload (key unit tests) 
            return projectEnrollment(
                currentSnapshotEnrollment,
                (double) historicalSnapshotEnrollment,
                (double) historicalDeadlineEnrollment);
        }

        /**
         * Performs the core projection computation over exact numeric values.
         * 
         * @param currentSnapshotEnrollment
         * @param historicalSnapshotEnrollment
         * @param historicalDeadlineEnrollment
         * @return the projected enrollment, rounded to the nearest student
         */
        public static int projectEnrollment(
            int currentSnapshotEnrollment,
            double historicalSnapshotEnrollment,
            double historicalDeadlineEnrollment) {
            requirePositiveEnrollmentValue(currentSnapshotEnrollment,
                "Current enrollment must be nonnegative.");
            requirePositiveEnrollmentValue(historicalSnapshotEnrollment,
                "Historical snapshot enrollment must be nonnegative.");
            requirePositiveEnrollmentValue(historicalDeadlineEnrollment,
                "Historical deadline enrollment must be nonnegative.");

            if (historicalSnapshotEnrollment == 0) {
                return projectWithoutHistoricalGrowth(
                    currentSnapshotEnrollment, historicalDeadlineEnrollment);
            }

            return roundToNearestStudent(
                currentSnapshotEnrollment
                    * (double) historicalDeadlineEnrollment
                    / historicalSnapshotEnrollment);
        }

        /**
         * Case where no one was enrolled at the historical
         * snapshot date: assume no change, unless history grew from zero.
         * 
         * @param currentSnapshotEnrollment
         * @param historicalDeadlineEnrollment
         * @return the projected enrollment without factoring in historical enrollment (currentSnapshotEnrollment)
         */
        private static int projectWithoutHistoricalGrowth(int currentSnapshotEnrollment, double historicalDeadlineEnrollment) {
            if (historicalDeadlineEnrollment > 0) {
                throw new IllegalArgumentException(
                    "Enrollment growth did not occur until after the relative cutoff date; no projection can be made.");
            }
            return currentSnapshotEnrollment;
        }

        /** Reject negative enrollment values with a caller-specific message.
         * Rejects NaN/±∞ — possible if interpolation ever produced trash.
         * 
         * @param value the value to check
         * @param message the message to display if value invalid
         */
        private static void requirePositiveEnrollmentValue(double value, String message) {
            if (!Double.isFinite(value) || value < 0) {
                throw new IllegalArgumentException(message);
            }
        }

        /** Round a raw projected enrollment to the nearest whole student.
         * 
         * @param rawProjection the projection value
         * @return the projection, rounded to the nearest student (whole number)
         */
        private static int roundToNearestStudent(double rawProjection) {
            return (int) Math.round(rawProjection);
        }
    }
}
