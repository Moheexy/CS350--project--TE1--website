package edu.odu.cs.cs350;

import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Objects;
import java.util.Set;

/**
 * 
 * Offering; an offering of a course. May consist of 1 or more sections
 */
public final class Offering {

    /** The name the course we offer. 
     * Only one name is employed, even for combined sections (e.g CS 450/550).
     */
    private final String CourseName;
    /**The overall enrollment for this offering. */
    private final int overallEnrollment;
    /** The overall cap of enrollment for this offering. */
    private final int overallEnrollmentCap;
    /**The sections in this offering. */
    private final ArrayList<Section> sections;

    /**
     * Constructor for offering.
     * 
     * @param potentialSections of this offering
     * @throws IllegalArgumentException if invariants of sections are violated
     */
    public Offering(List<Section> potentialSections) throws IllegalArgumentException
    {
        //If sections are invalid, we cannot proceed and want to throw.
        checkSectionsValid(potentialSections);
        
        this.sections = new ArrayList<>(potentialSections);
        this.CourseName = sections.get(0).subject() + sections.get(0).CourseNumber();
       /**
        * (overallEnrollmentCap/overallEnrollment) straight from the CSV; every seat belongs to lectures
        * (lecture + lab/rec) the totals include lab/rec seats, so instead sum each lecture section's own cap (XLSTCAP) and enrollment (ENR). 
        * A lecture-only-in-a-mixed-offering therefore reports only its lecture numbers; an all-lab/rec offering reports 0/0,
        * 
        * Note: Because labs and lectures are supposed to get filtered by ProjectionReader, lectureOnly should always be true for real data.
        * However, this is still useful to ensure Offerings/Sections created for testing purposes are legitimate.
       **/
       boolean lectureOnly = sections.stream().allMatch(Section::isLecture);
        this.overallEnrollmentCap = lectureOnly
            ? sections.get(0).overallEnrollmentCap()
            : sections.stream().filter(Section::isLecture).mapToInt(Section::CrossListCap).sum();
        this.overallEnrollment = lectureOnly
            ? sections.get(0).overallEnrollment()
            : sections.stream().filter(Section::isLecture).mapToInt(Section::ENR).sum();
        
    }


    /**
     * Checks the sections of our potential offering for validity.
     *
     *  
     * @param sectionsList the sections to check and assert invariants upon
     * @throws IllegalArgumentException if invariants are not held
     */
    private void checkSectionsValid(List<Section> sectionsList) throws IllegalArgumentException {

            enforceMoreThanTwoSections(sectionsList);
    
            String subject = sectionsList.get(0).subject();
            String crossListGroup = sectionsList.get(0).crossListGroup();
            int overallEnrollmentCapToCheck = sectionsList.get(0).overallEnrollmentCap();
            int overallEnrollmentToCheck = sectionsList.get(0).overallEnrollment();

            Set<Integer> knownCourseRegistrationNumbers = new HashSet<>();
            int realTotalEnrollment = 0;

            
        for (int sectionIndex = 0; sectionIndex < sectionsList.size(); sectionIndex++) {
                
            Section sectionToCheck = sectionsList.get(sectionIndex);

            enforceNoSoleSectionInMultiSectionOffering(sectionsList, sectionToCheck);

            checkDuplicateCourseRegistrationNumbers(knownCourseRegistrationNumbers, sectionToCheck);



            enforceSubjectsMatch(subject, sectionToCheck);

                
            enforceCrossListGroupsMatch(crossListGroup, sectionToCheck);


            enforceOverallEnrollmentCapMatches(overallEnrollmentCapToCheck, sectionToCheck);

            enforceOverallEnrollmentMatches(overallEnrollmentToCheck, sectionToCheck);
            
            realTotalEnrollment+= sectionToCheck.ENR();            
        }

        if(realTotalEnrollment != overallEnrollmentToCheck)
        {
            throw new IllegalArgumentException("The sum of individual enrollments must match the overall enrollment all entries in a section.");
        }

        //This invariant does not hold in real data
        // if(realTotalEnrollment > overallEnrollmentCap)
        // {
        //     throw new IllegalArgumentException(enrollmentExceedsTotal);

        // }
    }


    private void enforceOverallEnrollmentMatches(int overallEnrollmentToCheck, Section sectionToCheck) {
        if(overallEnrollmentToCheck != sectionToCheck.overallEnrollment())
        {
            throw new IllegalArgumentException("Overall enrollment must match for all entries in a section.");

        }
    }


    private void enforceOverallEnrollmentCapMatches(int overallEnrollmentCapToCheck, Section sectionToCheck) {
        if(overallEnrollmentCapToCheck != sectionToCheck.overallEnrollmentCap())
        {
            throw new IllegalArgumentException("The overall enrollment cap must match for all entries in a section.");

        }
    }


    private void enforceCrossListGroupsMatch(String crossListGroup, Section sectionToCheck) {
        if(!crossListGroup.equals(sectionToCheck.crossListGroup()))
        {
            throw new IllegalArgumentException("Cross-list group must match for all entries in a section.");

        }
    }


    private void enforceSubjectsMatch(String subject, Section sectionToCheck) {
        if(!subject.equals(sectionToCheck.subject()))
        {
            throw new IllegalArgumentException("Subject must match for all entries in a section.");
        }
    }


    private void checkDuplicateCourseRegistrationNumbers(Set<Integer> knownCourseRegistrationNumbers, Section sectionToCheck) {
        if(knownCourseRegistrationNumbers.contains(sectionToCheck.CRN()))
        {
          throw new IllegalArgumentException("CRN must be unique for all sections.");  
        }
        
        knownCourseRegistrationNumbers.add(sectionToCheck.CRN());
    }


    private void enforceNoSoleSectionInMultiSectionOffering(List<Section> sectionsToCheck, Section sectionToCheck) {
        if(sectionToCheck.crossListGroup().equals("None") && sectionsToCheck.size() > 1)
        {
            throw new IllegalArgumentException("If a section has no cross list group, it must be the only section in its offering.");
        }
    }


    private void enforceMoreThanTwoSections(List<Section> sectionsToCheck) {
        if(sectionsToCheck.size()<1)
        {
            throw new IllegalArgumentException("An offering must have at least one section.");
        }
    }


    /**
     * 
     * @return the sections of this offering
     */
    public ArrayList<Section> getSections() {
        return sections;
    }

    /**
     * 
     * @return the course of this offering. Even for mixed graduate/undergrad classes only one coursename is permitted
     */
    public String getCourseName() {
        return CourseName;
    }

    /**
     * 
     * @return overall enrollment in this offering
     */
    public int getOverallEnrollmentValue() {
        return overallEnrollment;
    }

    /**
     * 
     * @return overall cap of this offering
     */
    public int getOverallEnrollmentValueCap() {
        return overallEnrollmentCap;
    }

    /**
     * 
     * Inspiration: https://samderlust.com/dev-blog/java/overriding-equals-method-java.
     * @see java.lang.Object#equals(java.lang.Object)
     */
    @Override 
    public boolean equals(Object o)
    {
        if (exactMatch(o)) return true;

        if (nullOrMismatchedClass(o)) return false;

        Offering other = (Offering) o;
        boolean match = other.getCourseName().equals(this.getCourseName()) && //An offering can be made up of multiple sections of different course names
                        other.getOverallEnrollmentValueCap() == this.getOverallEnrollmentValueCap() &&
                        other.getOverallEnrollmentValue() == this.getOverallEnrollmentValue();

        for(Section section : other.getSections())
        {
            if(!this.getSections().contains(section))
            {
                match = false;

            }
        }
        return match;
    }

    private boolean nullOrMismatchedClass(Object o) {
        if (o == null || this.getClass() != o.getClass()) {
            return true;
        }
        return false;
    }

    private boolean exactMatch(Object o) {
        if (this == o) {
            return true;
        }
        return false;
    }

    /**
     * 
     * @see java.lang.Object#hashCode()
     */
    @Override 
    public int hashCode()
    {
        return Objects.hash(this.getCourseName(),this.getOverallEnrollmentValueCap(), this.getOverallEnrollmentValue(), this.getSections());
    }
    
}
